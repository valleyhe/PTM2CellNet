# CNN/Transformer E2E loop:
#     pip install -r requirements-core.txt
#     python scripts/train.py --config configs/smoke/cnn_cpu.yaml ...
#     python scripts/predict.py --model outputs/.../best_model.pt ...

# Heavy / CUDA-sensitive capabilities (pretrained LMs, Mamba, scVI/analysis)
# live in their own requirements-*.txt files and are NOT pulled in by default.
